cmake_minimum_required(VERSION 3.20)
project(claro_offline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(CTRANSLATE2_INCLUDE ctranslate2/translator.h REQUIRED)
find_library(CTRANSLATE2_LIBRARY NAMES ctranslate2 ctranslate2.4.8.4 REQUIRED)
find_path(SENTENCEPIECE_INCLUDE sentencepiece_processor.h REQUIRED)
find_library(SENTENCEPIECE_LIBRARY NAMES sentencepiece REQUIRED)
add_executable(claro-offline runner.cpp)
target_include_directories(claro-offline PRIVATE ${CTRANSLATE2_INCLUDE} ${SENTENCEPIECE_INCLUDE})
target_link_libraries(claro-offline PRIVATE ${CTRANSLATE2_LIBRARY} ${SENTENCEPIECE_LIBRARY})
if(APPLE)
  set_target_properties(claro-offline PROPERTIES INSTALL_RPATH "@executable_path")
endif()
install(TARGETS claro-offline DESTINATION .)
